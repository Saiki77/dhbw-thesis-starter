% Writing rule for all chapters: ONE SENTENCE PER LINE.
% The PDF preview jumps to source lines (SyncTeX); only then does it land
% exactly on the sentence under the cursor. A blank line starts a new paragraph.

\chapter{Introduction}\label{ch:introduction}

\section{Problem Statement}\label{sec:problem}

Describe here which problem the work solves and why it matters.
Each sentence goes on its own line.

\section{Objective}\label{sec:objective}

State the research question or the goal of the work here.

\section{Structure of the Work}\label{sec:structure}

\autoref{ch:background} explains the background.
\autoref{ch:method} describes the approach and \autoref{ch:implementation} the implementation.
\autoref{ch:results} presents the results, \autoref{ch:conclusion} sums them up.
